# Hero 3D Drop Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace static DALL-E amber drop image with interactive 3D glass teardrop that tracks cursor and refracts the fluid background.

**Architecture:** New module `drop-3d.js` renders a glTF teardrop on a transparent `<canvas>` overlaid on the fluid background. The drop uses a custom refraction shader that samples the fluid-bg canvas as an environment map. Mouse/gyroscope input tilts the drop ±15°. DALL-E image remains as fallback for no-WebGL devices.

**Tech Stack:** Three.js 0.177.0 (same version as fluid-bg.js), GLTFLoader, custom refraction ShaderMaterial, Blender MCP for glTF export.

- [ ] **Step 1: Generate clean teardrop in Blender via MCP**

Use Blender MCP to create a smooth teardrop mesh (subdiv 3, ~5K verts). Remove all materials — shader will be applied in Three.js.

```python
import bpy, math

# Clear scene
bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete()

# Create UV sphere base
bpy.ops.mesh.primitive_uv_sphere_add(segments=64, ring_count=48, radius=1)
obj = bpy.context.active_object
obj.name = 'AmberDrop'

# Shape into teardrop via proportional editing lattice
# Use mesh edit: scale top vertices to taper
bpy.ops.object.mode_set(mode='EDIT')
bpy.ops.mesh.select_all(action='DESELECT')
import bmesh
bm = bmesh.from_edit_mesh(obj.data)
for v in bm.verts:
    if v.co.z > 0:
        # Taper: higher z = narrower
        t = v.co.z  # 0..1
        scale = max(0.01, 1.0 - t**0.6 * 0.95)
        v.co.x *= scale
        v.co.y *= scale
        # Stretch up
        v.co.z = v.co.z ** 0.5 * 1.8
    else:
        # Bottom: rounder
        v.co.z *= 0.8
bmesh.update_edit_mesh(obj.data)
bpy.ops.object.mode_set(mode='OBJECT')

# Smooth shading
bpy.ops.object.shade_smooth()

# Remove all materials
obj.data.materials.clear()

# Export glTF
bpy.ops.export_scene.gltf(
    filepath='/Users/vladimir/amberra/models/amber-drop.glb',
    export_format='GLB',
    use_selection=True,
    export_apply=True,
    export_normals=True,
    export_materials='NONE'
)
print("Exported amber-drop.glb")
```

- [ ] **Step 3: Create drop-3d.js with refraction shader**

```javascript
// drop-3d.js — Interactive 3D amber teardrop with refraction
import * as THREE from 'https://esm.sh/three@0.177.0'
import { GLTFLoader } from 'https://esm.sh/three@0.177.0/examples/jsm/loaders/GLTFLoader.js'

const canvas = document.getElementById('drop-canvas')
if (!canvas) throw new Error('drop-canvas not found')

// Check WebGL support — show fallback image if not
const gl = canvas.getContext('webgl2') || canvas.getContext('webgl')
if (!gl) {
  canvas.style.display = 'none'
  const wrap = document.createElement('div')
  wrap.className = 'hero-drop-wrap'
  wrap.innerHTML = '<img src="images/amber-drop-hero.webp?v=10" alt="Amber drop" class="hero-drop" loading="eager">'
  canvas.parentElement.appendChild(wrap)
  throw new Error('WebGL not supported, fallback shown')
}

const renderer = new THREE.WebGLRenderer({ canvas, alpha: true, antialias: true })
renderer.setSize(innerWidth, innerHeight)
renderer.setPixelRatio(Math.min(devicePixelRatio, 2))
renderer.setClearColor(0x000000, 0) // transparent

const scene = new THREE.Scene()
const camera = new THREE.PerspectiveCamera(45, innerWidth / innerHeight, 0.1, 100)
camera.position.set(0, 0, 4)

// Capture fluid-bg canvas as environment texture
const fluidCanvas = document.getElementById('fluid-canvas')
const bgTexture = new THREE.CanvasTexture(fluidCanvas)
bgTexture.minFilter = THREE.LinearFilter
bgTexture.magFilter = THREE.LinearFilter

// Custom refraction material
const refractionMaterial = new THREE.ShaderMaterial({
  uniforms: {
    uBgTexture: { value: bgTexture },
    uTime: { value: 0 },
    uMouse: { value: new THREE.Vector2(0.5, 0.5) },
    uIOR: { value: 1.55 },
    uColor: { value: new THREE.Color(0.72, 0.53, 0.15) }, // golden amber
    uOpacity: { value: 0.92 },
  },
  vertexShader: `
    varying vec3 vNormal;
    varying vec3 vViewDir;
    varying vec2 vUv;
    varying vec3 vWorldPos;

    void main() {
      vUv = uv;
      vec4 worldPos = modelMatrix * vec4(position, 1.0);
      vWorldPos = worldPos.xyz;
      vNormal = normalize(normalMatrix * normal);
      vViewDir = normalize(cameraPosition - worldPos.xyz);
      gl_Position = projectionMatrix * viewMatrix * worldPos;
    }
  `,
  fragmentShader: `
    precision highp float;

    uniform sampler2D uBgTexture;
    uniform float uTime;
    uniform vec2 uMouse;
    uniform float uIOR;
    uniform vec3 uColor;
    uniform float uOpacity;

    varying vec3 vNormal;
    varying vec3 vViewDir;
    varying vec2 vUv;
    varying vec3 vWorldPos;

    void main() {
      vec3 N = normalize(vNormal);
      vec3 V = normalize(vViewDir);

      // Refraction: sample background through drop
      vec3 refracted = refract(-V, N, 1.0 / uIOR);
      vec2 refUV = gl_FragCoord.xy / vec2(${innerWidth}.0, ${innerHeight}.0);
      refUV += refracted.xy * 0.08;
      vec3 bgColor = texture2D(uBgTexture, refUV).rgb;

      // Fresnel: more reflection at edges
      float fresnel = pow(1.0 - max(dot(N, V), 0.0), 3.0);

      // Specular highlights tracking mouse
      vec3 lightDir = normalize(vec3(uMouse.x * 2.0 - 1.0, uMouse.y * 2.0 - 1.0, 1.5));
      vec3 H = normalize(lightDir + V);
      float spec = pow(max(dot(N, H), 0.0), 64.0);
      float spec2 = pow(max(dot(N, H), 0.0), 16.0) * 0.3;

      // Combine: refracted bg + amber tint + fresnel reflection + specular
      vec3 tinted = mix(bgColor, uColor, 0.45);
      vec3 reflection = vec3(0.9, 0.75, 0.4) * fresnel * 0.6;
      vec3 specular = vec3(1.0, 0.92, 0.7) * (spec + spec2);

      vec3 final = tinted + reflection + specular;

      // Subtle inner glow
      float rim = 1.0 - max(dot(N, V), 0.0);
      final += uColor * rim * 0.15;

      gl_FragColor = vec4(final, uOpacity + fresnel * 0.08);
    }
  `,
  transparent: true,
  side: THREE.FrontSide,
})

// Load glTF model
let dropMesh = null
const loader = new GLTFLoader()
loader.load('/models/amber-drop.glb', (gltf) => {
  dropMesh = gltf.scene.children[0]
  dropMesh.material = refractionMaterial
  dropMesh.scale.setScalar(0.9) // adjust to ~18vw visual size
  scene.add(dropMesh)
}, undefined, (err) => {
  console.error('Failed to load drop model:', err)
  // Show fallback
  canvas.style.display = 'none'
  const wrap = document.createElement('div')
  wrap.className = 'hero-drop-wrap'
  wrap.innerHTML = '<img src="images/amber-drop-hero.webp?v=10" alt="Amber drop" class="hero-drop" loading="eager">'
  canvas.parentElement.appendChild(wrap)
})

// Mouse tracking for specular + tilt
const mouse = { x: 0.5, y: 0.5, targetX: 0.5, targetY: 0.5 }
window.addEventListener('mousemove', (e) => {
  mouse.targetX = e.clientX / innerWidth
  mouse.targetY = 1.0 - e.clientY / innerHeight
})

// Mobile: gyroscope
if (window.DeviceOrientationEvent) {
  window.addEventListener('deviceorientation', (e) => {
    if (e.gamma !== null) {
      mouse.targetX = 0.5 + (e.gamma / 90) * 0.5 // -90..90 -> 0..1
      mouse.targetY = 0.5 + (e.beta - 45) / 90 * 0.5  // tilt
    }
  })
}

// Animation loop
const clock = new THREE.Clock()
function animate() {
  requestAnimationFrame(animate)
  const t = clock.getElapsedTime()

  // Smooth mouse lerp
  mouse.x += (mouse.targetX - mouse.x) * 0.05
  mouse.y += (mouse.targetY - mouse.y) * 0.05

  if (dropMesh) {
    // Float animation (same as CSS was doing)
    dropMesh.position.y = Math.sin(t * 1.57) * 0.08 // ~4s cycle

    // Tilt toward cursor ±15 degrees
    const tiltX = (mouse.y - 0.5) * 0.52 // ±15° in radians
    const tiltY = (mouse.x - 0.5) * 0.52
    dropMesh.rotation.x = tiltX
    dropMesh.rotation.y = tiltY

    // Slow idle rotation
    dropMesh.rotation.z = Math.sin(t * 0.5) * 0.03
  }

  // Update uniforms
  refractionMaterial.uniforms.uTime.value = t
  refractionMaterial.uniforms.uMouse.value.set(mouse.x, mouse.y)

  // Update bg texture from fluid canvas every frame
  bgTexture.needsUpdate = true

  renderer.render(scene, camera)
}
animate()

// Resize
window.addEventListener('resize', () => {
  renderer.setSize(innerWidth, innerHeight)
  camera.aspect = innerWidth / innerHeight
  camera.updateProjectionMatrix()
})
```
